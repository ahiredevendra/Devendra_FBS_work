Question 1 : Write a program to count number of words in a string. 

int mystrlen(char*)
void main()
{
	char str[20] = "devendra";

	int x = mystrlen(str);
	printf("Length of string is : %d",x);
}
int mystrlen(char* str)
{
	int count;
	
	int i = 0;

	while(str[i]!='\0')
	{
		count++;
		i++;
	}

	return count;
}
